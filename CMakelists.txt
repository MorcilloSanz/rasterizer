cmake_minimum_required(VERSION 3.15)

project(rasterizer
    VERSION 1.0.0
    DESCRIPTION "Software rasterizer library"
    LANGUAGES C
)

# C11, no compiler extensions (equivalent to -std=c11)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

# Only build the test program when this is the top-level project
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    set(RASTERIZER_IS_TOP_LEVEL ON)
else()
    set(RASTERIZER_IS_TOP_LEVEL OFF)
endif()

option(RASTERIZER_BUILD_TEST "Build the test program" ${RASTERIZER_IS_TOP_LEVEL})
option(BUILD_SHARED_LIBS "Build rasterizer as a shared library" OFF)

# ----------------------------------------------------------------------------
# Library: rasterizer
# ----------------------------------------------------------------------------
add_library(rasterizer rasterizer/rasterizer.c)
add_library(rasterizer::rasterizer ALIAS rasterizer)

# The project root is used as the include path (equivalent to -I.),
# so consumers can write #include "rasterizer/rasterizer.h"
target_include_directories(rasterizer
    PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
)

# stb_truetype and friends need libm on Linux/macOS
if(NOT WIN32)
    target_link_libraries(rasterizer PUBLIC m)
endif()

if(MSVC)
    target_compile_options(rasterizer PRIVATE /W3)
else()
    target_compile_options(rasterizer PRIVATE -Wall)
endif()

# ----------------------------------------------------------------------------
# Test program
# ----------------------------------------------------------------------------
if(RASTERIZER_BUILD_TEST)
    add_subdirectory(test)
endif()
